Write a program to find the largest and smallest element in an array

Description: Finds both minimum and maximum values from an array of numbers.

Working Implementation & Output

Output will appear here.

function getMinMax(arr) {
  let min = arr[0], max = arr[0];
  for (let num of arr) {
    if (num < min) min = num;
    if (num > max) max = num;
  }
  return { min, max };
}

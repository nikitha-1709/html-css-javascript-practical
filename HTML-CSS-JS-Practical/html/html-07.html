Source Code

<iframe 
  src="html-01.html" 
  width="100%" 
  height="300" 
  title="Embedded Experiment" 
  loading="lazy"
  style="border: 1px solid #ccc;">
</iframe>

<!-- Linking to Frame Target -->
<a href="html-01.html" target="myFrame">Show Page 1</a>
<iframe name="myFrame" src="html-01.html"></iframe>
